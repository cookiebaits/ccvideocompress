import { FFmpeg } from 'https://unpkg.com/@ffmpeg/ffmpeg@0.12.6/dist/esm/index.js'; import { fetchFile } from 'https://unpkg.com/@ffmpeg/util@0.12.1/dist/esm/index.js'; const ffmpeg = new FFmpeg(); const uploader = document.getElementById('uploader'); const convertBtn = document.getElementById('convert-btn'); const message = document.getElementById('message'); const progressContainer = document.getElementById('progress-container'); const progressBar = document.getElementById('progress-bar'); const statusText = document.getElementById('status-text'); const downloadLink = document.getElementById('download-link'); const trimStartInput = document.getElementById('trim-start'); const trimEndInput = document.getElementById('trim-end'); let currentObjectUrl = null; const cleanupMemory = () => { if (currentObjectUrl) { URL.revokeObjectURL(currentObjectUrl); currentObjectUrl = null; } try { ffmpeg.deleteFile('input.mp4'); ffmpeg.deleteFile('output.mp4'); } catch (e) { // Ignore if files haven't been created yet } }; window.addEventListener('beforeunload', cleanupMemory); const getVideoDuration = (file) => { return new Promise((resolve) => { const video = document.createElement('video'); video.preload = 'metadata'; video.onloadedmetadata = function() { URL.revokeObjectURL(video.src); resolve(video.duration); } video.src = URL.createObjectURL(file); }); }; // --- NEW CACHING SYSTEM START --- const getCachedFFmpegFile = async (url, type) => { const cacheName = 'ffmpeg-engine-cache-v1'; const cache = await caches.open(cacheName); let response = await cache.match(url); if (response) { console.log(`Loaded FFmpeg ${type} from local cache.`); } else { console.log(`Downloading FFmpeg ${type} for the first time...`); message.innerText = `Downloading FFmpeg engine (${type})... this only happens once!`; response = await fetch(url); if (response.ok) { // Save a copy to the cache for next time await cache.put(url, response.clone()); } else { throw new Error(`Failed to download ${type}`); } } // Convert the cached response into a local blob URL const blob = await response.blob(); return URL.createObjectURL(blob); }; // --- NEW CACHING SYSTEM END --- const loadFFmpeg = async () => { message.innerText = "Checking cache for FFmpeg engine..."; try { // Fetch files through our custom cache function const coreURL = 'https://unpkg.com/@ffmpeg/core@0.12.4/dist/esm/ffmpeg-core.js'; const wasmURL = 'https://unpkg.com/@ffmpeg/core@0.12.4/dist/esm/ffmpeg-core.wasm'; const cachedCoreURL = await getCachedFFmpegFile(coreURL, 'Core JS'); const cachedWasmURL = await getCachedFFmpegFile(wasmURL, 'WASM'); ffmpeg.on('progress', ({ progress }) => { const percent = Math.min(Math.max(Math.round(progress * 100), 0), 100); progressBar.value = percent; statusText.innerText = `Processing... ${percent}%`; }); // Load FFmpeg using the local cached Blob URLs await ffmpeg.load({ coreURL: cachedCoreURL, wasmURL: cachedWasmURL, }); message.innerText = "Ready!"; convertBtn.innerText = "Compress & Trim Video"; convertBtn.disabled = false; // Clean up the temporary Blob URLs used for loading URL.revokeObjectURL(cachedCoreURL); URL.revokeObjectURL(cachedWasmURL); } catch (error) { console.error(error); message.innerText = "Error loading FFmpeg engine. Please check your connection and refresh."; } }; const processVideo = async () => { const file = uploader.files[0]; if (!file) { alert("Please select a video file first."); return; } cleanupMemory(); downloadLink.style.display = "none"; const trimStart = parseFloat(trimStartInput.value) || 0; const trimEnd = parseFloat(trimEndInput.value) || 0; const totalDuration = await getVideoDuration(file); const durationToKeep = totalDuration - trimStart - trimEnd; if (durationToKeep <= 0) { alert("Your trim times remove the entire video! Please adjust the seconds."); return; } convertBtn.disabled = true; progressContainer.style.display = "block"; message.innerText = "Reading file into browser memory..."; await ffmpeg.writeFile('input.mp4', await fetchFile(file)); message.innerText = "Processing... Leave this tab open."; const ffmpegArgs = []; if (trimStart > 0) { ffmpegArgs.push('-ss', String(trimStart)); } if (trimStart > 0 || trimEnd > 0) { ffmpegArgs.push('-t', String(durationToKeep)); } ffmpegArgs.push( '-i', 'input.mp4', '-c:v', 'libx264', '-crf', '23', '-preset', 'veryfast', '-c:a', 'aac', '-b:a', '128k', '-movflags', '+faststart', 'output.mp4' ); await ffmpeg.exec(ffmpegArgs); message.innerText = "Finalizing file..."; const data = await ffmpeg.readFile('output.mp4'); currentObjectUrl = URL.createObjectURL(new Blob([data.buffer], { type: 'video/mp4' })); downloadLink.href = currentObjectUrl; downloadLink.style.display = "block"; message.innerText = "Done! Click the link above to save."; progressContainer.style.display = "none"; convertBtn.disabled = false; }; convertBtn.addEventListener('click', processVideo); loadFFmpeg();
